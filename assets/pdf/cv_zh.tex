\section{学术成果}

\pubentry{When Slower Isn't Truer: Inverse Scaling Law of Truthfulness in Multimodal Reasoning}{\venue{ACL 2026}}{https://arxiv.org/abs/2505.20214}
\me{}, Wenjing Cao, Jiahao Li, Xuyao Wang, Chi-Min Chan, Sirui Han, Juntao Dai, Yike Guo, Yaodong Yang, Jiaming Ji
\begin{itemize}
  \item 发现逆缩放定律：更慢的推理模型在多模态场景中反而更不真实。
  \item 提出 \textbf{TruthfulVQA}，首个多模态真实性评测基准（5,000+图像），以及 \textbf{TruthfulJudge} 人机协同评测框架。
\end{itemize}

\vspace{0.3em}

\pubentry{Debate with Images: Detecting Deceptive Behaviors in Multimodal Large Language Models}{\venue{ICML 2026}}{https://arxiv.org/abs/2512.00349}
\me{}, Shiyi Hou, Kaile Wang, Boyuan Chen, Donghai Hong, Jiayi Zhou, Juntao Dai, Yaodong Yang, Jiaming Ji
\begin{itemize}
  \item 提出 \textbf{MM-DeceptionBench}，首个多模态大模型欺骗行为评测基准。
  \item 提出 \textbf{Debate with Images}，基于视觉证据的多智能体辩论框架，显著提升欺骗检测准确率。
\end{itemize}

\vspace{0.3em}

\pubentry{AI Alignment Can Build Moral Autonomy}{\venue{NeurIPS 2026}}{https://openreview.net/forum?id=htzFr3tI5z}
Kaile Wang*, Hantao Lou*, Tianyi Qiu*, Sebastian Sunday-Grève, \me{}, Xinxiong Wu, Jiaming Ji, Yaodong Yang
\hfill {\small *\,共同贡献}
\begin{itemize}
  \item 探讨以道德自主性为对齐目标，强调反思性认可、明确的原则与实践智慧。
\end{itemize}

\vspace{0.3em}

\pubentry{A Blind Spot in Alignment: Quantifying Biosecurity Risks in Large Language Models}{\venue{COLM 2026}}{https://arxiv.org/abs/2608.02684}
Shu Quan, Tianfang Hao, \me{}, He Geng, Jiayi Zhou, Boyuan Chen, Kaile Wang, Donghai Hong, Juntao Dai, Yaodong Yang, Jiaming Ji

\end{itemize}
